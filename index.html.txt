    function checkPassword() {

        const input = document.getElementById('passwordInput').value.trim();

        const correctPassword = 'I LOVE YOU';

        const errorDiv = document.getElementById('errorMessage');

        if (input.toUpperCase() === correctPassword.toUpperCase()) {

            // Correct password

            errorDiv.innerHTML = '';

            unlockSurprise();

        } else {

            // Wrong password

            errorDiv.innerHTML = '❌ Oops! That\'s not right... Try again! 😊';

            document.getElementById('passwordInput').value = '';

            document.getElementById('passwordInput').shake = true;

        }

    }

    // ==================== COUNTDOWN TIMER ====================

    function startCountdown() {

        function updateCountdown() {

            // Calculate next Eid (assuming next Eid is in the future)

            // Using June 15, 2026 as example Eid date (adjust as needed)

            const eidDate = new Date('2026-06-16').getTime();

            const now = new Date().getTime();

            const distance = eidDate - now;

            if (distance > 0) {

                const days = Math.floor(distance / (1000 * 60 * 60 * 24));

                const hours = Math.floor((distance % (1000 * 60 * 60 * 24)) / (1000 * 60 * 60));

                const minutes = Math.floor((distance % (1000 * 60 * 60)) / (1000 * 60));

                const seconds = Math.floor((distance % (1000 * 60)) / 1000);

                document.getElementById('days').textContent = days;

                document.getElementById('hours').textContent = hours;

                document.getElementById('minutes').textContent = minutes;

                document.getElementById('seconds').textContent = seconds;

            }

        }

        updateCountdown();

        setInterval(updateCountdown, 1000);

    }

    // ==================== FIREWORKS ====================

    function triggerFireworks() {

        const fireworksContainer = document.createElement('div');

        fireworksContainer.className = 'fireworks';

        document.body.appendChild(fireworksContainer);

        for (let i = 0; i < 50; i++) {

            setTimeout(() => {

                createFirework(fireworksContainer);

            }, i * 100);

        }

    }

    function createFirework(container) {

        const x = Math.random() * window.innerWidth;

        const y = Math.random() * window.innerHeight * 0.6;

        const colors = ['#ff69b4', '#ffc0cb', '#ffd700', '#ff1493', '#ffb6c1'];

        for (let i = 0; i < 30; i++) {

            const firework = document.createElement('div');

            firework.className = 'firework';

            firework.style.left = x + 'px';

            firework.style.top = y + 'px';

            firework.style.backgroundColor = colors[Math.floor(Math.random() * colors.length)];

            const angle = (Math.PI * 2 * i) / 30;

            const velocity = 5 + Math.random() * 5;

            const tx = Math.cos(angle) * velocity;

            const ty = Math.sin(angle) * velocity;

            container.appendChild(firework);

            let x_pos = 0;

            let y_pos = 0;

            const animate = () => {

                x_pos += tx;

                y_pos += ty;

                y_pos += 0.2; // Gravity

                firework.style.transform = `translate(${x_pos}px, ${y_pos}px)`;

                firework.style.opacity = 1 - (Math.abs(y_pos) / 300);

                if (Math.abs(y_pos) < 300) {

                    requestAnimationFrame(animate);

                } else {

                    firework.remove();

                }

            };

            animate();

        }

    }

    function toggleMusic() {

        const audio = document.getElementById('backgroundMusic');

        const btn = document.getElementById('musicBtn');

        if (audio.paused) {

            audio.play().catch(err => {

                console.log('Music playback failed:', err);

                alert('Could not play music. Browser may require user interaction first.');

            });

            btn.textContent = '🎵 Stop Love Song';

            btn.style.opacity = '1';

        } else {

            audio.pause();

            audio.currentTime = 0;

            btn.textContent = '🎵 Play Love Song';

            btn.style.opacity = '0.7';

        }

    }

    // ==================== GAME ====================

    let gameScore = 0;

    const targetScore = 20;

    let gameActive = false;

    function startGame() {

        gameScore = 0;

        gameActive = true;

        document.getElementById('gameContainer').classList.add('show');

        document.getElementById('mainPage').classList.remove('show');

        document.getElementById('gameScore').textContent = '0';

        document.getElementById('winMessage').style.display = 'none';

        // Start spawning hearts

        const gameArea = document.getElementById('gameArea');

        gameArea.innerHTML = '';

        const spawnHearts = setInterval(() => {

            if (!gameActive) {

                clearInterval(spawnHearts);

                return;

            }

            createFallingHeart(gameArea);

        }, 600);

    }

    function createFallingHeart(container) {

        const heart = document.createElement('div');

        heart.className = 'falling-heart';

        heart.textContent = '❤️';

        heart.style.left = Math.random() * (window.innerWidth - 50) + 'px';

        heart.style.top = '-50px';

        const fallDuration = 4 + Math.random() * 2;

        heart.style.animation = `fall ${fallDuration}s linear`;

        heart.onclick = (e) => {

            e.stopPropagation();

            collectHeart(heart);

        };

        container.appendChild(heart);

        setTimeout(() => {

            if (heart.parentNode) {

                heart.remove();

            }

        }, fallDuration * 1000);

    }

    function collectHeart(heart) {

        gameScore++;

        document.getElementById('gameScore').textContent = gameScore;

        // Bounce effect

        heart.style.animation = 'none';

        heart.style.transform = 'scale(1.5)';

        heart.style.opacity = '0';

        // Fireworks on collection

        createMiniFirework(heart.offsetLeft, heart.offsetTop);

        setTimeout(() => heart.remove(), 300);

        if (gameScore >= targetScore) {

            endGame();

        }

    }

    function createMiniFirework(x, y) {

        const container = document.getElementById('gameArea');

        for (let i = 0; i < 15; i++) {

            const spark = document.createElement('div');

            spark.style.position = 'absolute';

            spark.style.left = x + 'px';

            spark.style.top = y + 'px';

            spark.style.fontSize = '1rem';

            spark.textContent = '✨';

            const angle = (Math.PI * 2 * i) / 15;

            const velocity = 3 + Math.random() * 3;

            const tx = Math.cos(angle) * velocity;

            const ty = Math.sin(angle) * velocity;

            container.appendChild(spark);

            let x_pos = 0;

            let y_pos = 0;

            const animate = () => {

                x_pos += tx;

                y_pos += ty;

                y_pos += 0.1;

                spark.style.transform = `translate(${x_pos}px, ${y_pos}px)`;

                spark.style.opacity = 1 - (Math.abs(y_pos) / 150);

                if (Math.abs(y_pos) < 150) {

                    requestAnimationFrame(animate);

                } else {

                    spark.remove();

                }

            };

            animate();

        }

    }

    function endGame() {

        gameActive = false;

        document.getElementById('gameArea').innerHTML = '';

        document.getElementById('winMessage').style.display = 'block';

        triggerFireworks();

    }

    }
